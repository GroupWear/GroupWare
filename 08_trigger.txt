DROP TRIGGER IF EXISTS TRG_LEAVE_BACKUP;
DROP TABLE IF EXISTS LEAVE_HISTORY_BACKUP;

CREATE TABLE LEAVE_HISTORY_BACKUP (
    BACKUP_ID INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    BACKUP_DATE TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    ACTION_TYPE VARCHAR(10),
    LEAVE_NO INT,
    EMP_NO INT,
    START_DATE DATE,
    END_DATE DATE,
    USE_DAYS INT,
    REASON VARCHAR(500),
    STATUS VARCHAR(50),
    APPROVAL_STEP INT,
    SIGN1 VARCHAR(50), SIGN1_DATE DATE,
    SIGN2 VARCHAR(50), SIGN2_DATE DATE,
    SIGN3 VARCHAR(50), SIGN3_DATE DATE,
    SIGN4 VARCHAR(50), SIGN4_DATE DATE,
    SIGN5 VARCHAR(50), SIGN5_DATE DATE
);

-- 앞서 만든 Java 클래스를 CALL 명령어로 연결 (에러 없이 즉시 생성됨)
CREATE TRIGGER TRG_LEAVE_BACKUP 
AFTER INSERT, UPDATE ON LEAVE_HISTORY 
FOR EACH ROW 
CALL "com.groupware.util.LeaveBackupTrigger";
