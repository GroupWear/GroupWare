-- 기존 테이블 및 시퀀스 삭제 (테스트 초기화용)
DROP TABLE BOARD_FILE CASCADE CONSTRAINTS;
DROP TABLE BOARD CASCADE CONSTRAINTS;
DROP SEQUENCE SEQ_BOARD;
DROP SEQUENCE SEQ_BOARD_FILE;

-- 게시판 테이블 생성 (사내 소식, 자유, 건의)
CREATE TABLE BOARD (
    BOARD_NO NUMBER PRIMARY KEY,
    BOARD_TYPE NUMBER NOT NULL, -- 1: 사내소식, 2: 자유, 3: 건의
    TITLE VARCHAR2(200) NOT NULL,
    CONTENT CLOB NOT NULL,
    EMP_NO NUMBER NOT NULL,
    REG_DATE DATE DEFAULT SYSDATE,
    HIT NUMBER DEFAULT 0,
    CONSTRAINT FK_BOARD_EMP FOREIGN KEY (EMP_NO) REFERENCES EMPLOYEE(EMP_NO)
);

-- 게시판 일련번호 시퀀스
CREATE SEQUENCE SEQ_BOARD START WITH 1 INCREMENT BY 1 NOCACHE;

-- 첨부파일 테이블 생성
CREATE TABLE BOARD_FILE (
    FILE_NO NUMBER PRIMARY KEY,
    BOARD_NO NUMBER NOT NULL,
    ORG_NAME VARCHAR2(300) NOT NULL,
    SAVED_NAME VARCHAR2(300) NOT NULL,
    FILE_SIZE NUMBER DEFAULT 0,
    CONSTRAINT FK_FILE_BOARD FOREIGN KEY (BOARD_NO) REFERENCES BOARD(BOARD_NO) ON DELETE CASCADE
);

-- 첨부파일 일련번호 시퀀스
CREATE SEQUENCE SEQ_BOARD_FILE START WITH 1 INCREMENT BY 1 NOCACHE;
