-- ==========================================
-- 0. 既存テーブル、シーケンス、トリガーの初期化 (削除)
-- ==========================================
-- トリガーおよびバックアップテーブルの削除
DROP TRIGGER IF EXISTS TRG_LEAVE_BACKUP;
DROP TABLE IF EXISTS LEAVE_HISTORY_BACKUP;
DROP TRIGGER IF EXISTS TRG_RENTAL_BACKUP;
DROP TABLE IF EXISTS RENTAL_HISTORY_BACKUP;

-- 制約条件を含むテーブルの削除
DROP TABLE BOARD_COMMENT CASCADE CONSTRAINTS;
DROP TABLE BOARD_FILE CASCADE CONSTRAINTS;
DROP TABLE BOARD CASCADE CONSTRAINTS;
DROP TABLE RENTAL_HISTORY CASCADE CONSTRAINTS;
DROP TABLE LEAVE_HISTORY CASCADE CONSTRAINTS;
DROP TABLE RESERVATION CASCADE CONSTRAINTS;
DROP TABLE EQUIPMENT CASCADE CONSTRAINTS;
DROP TABLE ROOM CASCADE CONSTRAINTS;
DROP TABLE EMPLOYEE CASCADE CONSTRAINTS;

-- シーケンスの削除
DROP SEQUENCE SEQ_COMMENT;
DROP SEQUENCE SEQ_BOARD_FILE;
DROP SEQUENCE SEQ_BOARD;
DROP SEQUENCE SEQ_RESERVATION;
DROP SEQUENCE SEQ_RENTAL;
DROP SEQUENCE SEQ_LEAVE;
DROP SEQUENCE SEQ_EQUIPMENT;

-- ==========================================
-- 1. シーケンス生成 (NOCACHE 適用)
-- ==========================================
CREATE SEQUENCE SEQ_RESERVATION START WITH 1 INCREMENT BY 1 NOCACHE;
CREATE SEQUENCE SEQ_RENTAL START WITH 1 INCREMENT BY 1 NOCACHE;
CREATE SEQUENCE SEQ_LEAVE START WITH 1 INCREMENT BY 1 NOCACHE;
CREATE SEQUENCE SEQ_EQUIPMENT START WITH 100 INCREMENT BY 1 NOCACHE;
CREATE SEQUENCE SEQ_BOARD START WITH 1 INCREMENT BY 1 NOCACHE;
CREATE SEQUENCE SEQ_BOARD_FILE START WITH 1 INCREMENT BY 1 NOCACHE;
CREATE SEQUENCE SEQ_COMMENT START WITH 1 INCREMENT BY 1 NOCACHE;

-- ==========================================
-- 2. テーブル生成 (正規化および制約条件含む)
-- ==========================================

-- 1. 社員テーブル (EMPLOYEE)
CREATE TABLE EMPLOYEE (
    EMP_NO NUMBER PRIMARY KEY,
    EMP_PW VARCHAR2(200),
    EMP_NAME VARCHAR2(50) NOT NULL,
    EMP_LEVEL NUMBER NOT NULL,
    MANAGER VARCHAR2(1) DEFAULT 'N',
    DEPT VARCHAR2(50),
    MAX_LEAVE NUMBER DEFAULT 15,
    CUR_LEAVE NUMBER DEFAULT 15,
    RETIRED VARCHAR2(1) DEFAULT 'N',
    TEMP_LEVEL NUMBER DEFAULT 0,
    CONSTRAINT CK_EMP_MANAGER CHECK (MANAGER IN ('Y', 'N')),
    CONSTRAINT CK_EMP_RETIRED CHECK (RETIRED IN ('Y', 'N'))
);

-- 2. 会議室テーブル (ROOM)
CREATE TABLE ROOM (
    ROOM_ID VARCHAR2(20) PRIMARY KEY,
    ROOM_NAME VARCHAR2(100) NOT NULL,
    CAPACITY NUMBER DEFAULT 0,
    HAS_BEAM VARCHAR2(1) DEFAULT 'N',
    DESCRIPTION VARCHAR2(500),
    ENABLE VARCHAR2(1) DEFAULT 'Y' NOT NULL,
    CONSTRAINT CK_ROOM_BEAM CHECK (HAS_BEAM IN ('Y', 'N')),
    CONSTRAINT CK_ROOM_ENABLE CHECK (ENABLE IN ('Y', 'N'))
);

-- 3. 備品マスタテーブル (EQUIPMENT)
CREATE TABLE EQUIPMENT (
    EQ_NO NUMBER PRIMARY KEY,
    EQ_NAME VARCHAR2(100) NOT NULL,
    TOTAL_COUNT NUMBER DEFAULT 0,
    REMAIN_COUNT NUMBER DEFAULT 0,
    ISCHECKED CHAR(1) DEFAULT 'Y' NOT NULL
);

-- 4. 予約テーブル (RESERVATION)
CREATE TABLE RESERVATION (
    RES_NO NUMBER PRIMARY KEY,
    EMP_NO NUMBER NOT NULL REFERENCES EMPLOYEE(EMP_NO),
    ROOM_ID VARCHAR2(20) NOT NULL REFERENCES ROOM(ROOM_ID),
    RES_DATE DATE NOT NULL,
    START_TIME VARCHAR2(10) NOT NULL,
    END_TIME VARCHAR2(10) NOT NULL,
    PURPOSE VARCHAR2(200),
    STATUS VARCHAR2(20) DEFAULT '予約完了',
    EXPIRE_TIME TIMESTAMP
);

-- 5. 備品貸出履歴テーブル (RENTAL_HISTORY)
CREATE TABLE RENTAL_HISTORY (
    RENTAL_NO NUMBER PRIMARY KEY,
    EMP_NO NUMBER NOT NULL REFERENCES EMPLOYEE(EMP_NO),
    EQ_NO NUMBER NOT NULL REFERENCES EQUIPMENT(EQ_NO),
    TITLE VARCHAR2(100),
    REQ_COUNT NUMBER DEFAULT 1,
    RENTAL_DATE DATE NOT NULL,
    RETURN_DATE DATE NOT NULL,
    STATUS VARCHAR2(20) DEFAULT '承認待ち',
    APPROVAL_STEP NUMBER DEFAULT 1,
    SIGN1 VARCHAR2(50), SIGN1_DATE DATE,
    SIGN2 VARCHAR2(50), SIGN2_DATE DATE,
    SIGN3 VARCHAR2(50), SIGN3_DATE DATE,
    SIGN4 VARCHAR2(50), SIGN4_DATE DATE,
    SIGN5 VARCHAR2(50), SIGN5_DATE DATE,
    REASON VARCHAR2(500)
);

-- 6. 休暇申請履歴テーブル (LEAVE_HISTORY)
CREATE TABLE LEAVE_HISTORY (
    LEAVE_NO NUMBER PRIMARY KEY,
    EMP_NO NUMBER NOT NULL REFERENCES EMPLOYEE(EMP_NO),
    START_DATE DATE NOT NULL,
    END_DATE DATE NOT NULL,
    USE_DAYS NUMBER NOT NULL,
    REASON VARCHAR2(500),
    STATUS VARCHAR2(20) DEFAULT '承認待ち',
    APPROVAL_STEP NUMBER DEFAULT 1,
    SIGN1 VARCHAR2(50), SIGN1_DATE DATE,
    SIGN2 VARCHAR2(50), SIGN2_DATE DATE,
    SIGN3 VARCHAR2(50), SIGN3_DATE DATE,
    SIGN4 VARCHAR2(50), SIGN4_DATE DATE,
    SIGN5 VARCHAR2(50), SIGN5_DATE DATE
);

-- 7. 掲示板テーブル (BOARD)
CREATE TABLE BOARD (
    BOARD_NO NUMBER PRIMARY KEY,
    BOARD_TYPE NUMBER NOT NULL, 
    TITLE VARCHAR2(200) NOT NULL,
    CONTENT CLOB NOT NULL,
    EMP_NO NUMBER NOT NULL,
    REG_DATE DATE DEFAULT SYSDATE,
    HIT NUMBER DEFAULT 0,
    CONSTRAINT FK_BOARD_EMP FOREIGN KEY (EMP_NO) REFERENCES EMPLOYEE(EMP_NO)
);

-- 8. 掲示板添付ファイルテーブル (BOARD_FILE)
CREATE TABLE BOARD_FILE (
    FILE_NO NUMBER PRIMARY KEY,
    BOARD_NO NUMBER NOT NULL,
    ORG_NAME VARCHAR2(300) NOT NULL,
    SAVED_NAME VARCHAR2(300) NOT NULL,
    FILE_SIZE NUMBER DEFAULT 0,
    CONSTRAINT FK_FILE_BOARD FOREIGN KEY (BOARD_NO) REFERENCES BOARD(BOARD_NO) ON DELETE CASCADE
);

-- 9. 掲示板コメントテーブル (BOARD_COMMENT)
CREATE TABLE BOARD_COMMENT (
    COMMENT_NO NUMBER PRIMARY KEY,
    BOARD_NO NUMBER NOT NULL,
    EMP_NO NUMBER NOT NULL,
    CONTENT VARCHAR2(1000) NOT NULL,
    REG_DATE DATE DEFAULT SYSDATE,
    PARENT_NO NUMBER, 
    GROUP_NO NUMBER NOT NULL, 
    ORDER_NO NUMBER DEFAULT 0, 
    DEPTH NUMBER DEFAULT 0, 
    CONSTRAINT FK_COMMENT_BOARD FOREIGN KEY (BOARD_NO) REFERENCES BOARD(BOARD_NO) ON DELETE CASCADE,
    CONSTRAINT FK_COMMENT_EMP FOREIGN KEY (EMP_NO) REFERENCES EMPLOYEE(EMP_NO),
    CONSTRAINT FK_COMMENT_PARENT FOREIGN KEY (PARENT_NO) REFERENCES BOARD_COMMENT(COMMENT_NO) ON DELETE CASCADE
);

-- 10. 休暇履歴バックアップテーブル (LEAVE_HISTORY_BACKUP)
CREATE TABLE LEAVE_HISTORY_BACKUP (
    BACKUP_ID INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    BACKUP_DATE TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    ACTION_TYPE VARCHAR(10),
    LEAVE_NO INT,
    EMP_NO INT,
    START_DATE DATE,
    END_DATE DATE,
    USE_DAYS INT,
    REASON VARCHAR(500),
    STATUS VARCHAR(50),
    APPROVAL_STEP INT,
    SIGN1 VARCHAR(50), SIGN1_DATE DATE,
    SIGN2 VARCHAR(50), SIGN2_DATE DATE,
    SIGN3 VARCHAR(50), SIGN3_DATE DATE,
    SIGN4 VARCHAR(50), SIGN4_DATE DATE,
    SIGN5 VARCHAR(50), SIGN5_DATE DATE
);

-- 11. 備品貸出履歴バックアップテーブル (RENTAL_HISTORY_BACKUP)
CREATE TABLE RENTAL_HISTORY_BACKUP (
    BACKUP_ID INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    BACKUP_DATE TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    ACTION_TYPE VARCHAR(10),
    RENTAL_NO INT,
    EMP_NO INT,
    EQ_NO INT,
    TITLE VARCHAR(200),
    REQ_COUNT INT,
    RENTAL_DATE DATE,
    RETURN_DATE DATE,
    STATUS VARCHAR(50),
    APPROVAL_STEP INT,
    SIGN1 VARCHAR(50), SIGN1_DATE DATE,
    SIGN2 VARCHAR(50), SIGN2_DATE DATE,
    SIGN3 VARCHAR(50), SIGN3_DATE DATE,
    SIGN4 VARCHAR(50), SIGN4_DATE DATE,
    SIGN5 VARCHAR(50), SIGN5_DATE DATE,
    REASON VARCHAR(500)
);

-- ==========================================
-- 3. インデックスおよびトリガー生成
-- ==========================================

-- 掲示板コメントインデックス
CREATE INDEX IDX_COMMENT_BOARD ON BOARD_COMMENT(BOARD_NO);
CREATE INDEX IDX_COMMENT_GROUP ON BOARD_COMMENT(GROUP_NO);

-- 休暇履歴バックアップトリガー
CREATE TRIGGER TRG_LEAVE_BACKUP 
AFTER INSERT, UPDATE ON LEAVE_HISTORY 
FOR EACH ROW 
CALL "com.groupware.util.LeaveBackupTrigger";

-- 備品貸出履歴バックアップトリガー
CREATE TRIGGER TRG_RENTAL_BACKUP 
AFTER INSERT, UPDATE ON RENTAL_HISTORY 
FOR EACH ROW 
CALL "com.groupware.util.RentalBackupTrigger";

COMMIT;
